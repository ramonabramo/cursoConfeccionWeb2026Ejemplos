    💡 ¿Qué hace la palabra reservada this en este ejemplo?

    const cuentaBancaria = {
  titular: "Lucía Navarro",
  saldo: 200,
  ingresar(cantidad) {
    this.saldo += cantidad; // 'this' hace referencia directa a este objeto cuentaBancaria
    return `Ingreso de ${cantidad}€. Saldo actual: ${this.saldo}€`;
  }
};
